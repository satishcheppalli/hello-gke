================================================================================
COMPLETE DEPLOYMENT GUIDE: Docker Build, GCR Push, and Kubernetes Deployment
================================================================================

This file contains all commands needed to build a Docker image, push it to
Google Container Registry (GCR), and deploy to Google Kubernetes Engine (GKE).

================================================================================
SECTION 1: PREREQUISITES & AUTHENTICATION
================================================================================

# 1.1 Authenticate with Google Cloud Platform
gcloud auth login

# 1.2 Set your GCP project
gcloud config set project YOUR_PROJECT_ID

# 1.3 Verify project is set
gcloud config get-value project

# 1.4 Configure Docker authentication with GCR
gcloud auth configure-docker

# 1.5 Enable required GCP APIs
gcloud services enable containerregistry.googleapis.com
gcloud services enable container.googleapis.com
gcloud services enable artifactregistry.googleapis.com


================================================================================
SECTION 2: BUILD DOCKER IMAGE
================================================================================

# 2.1 Set environment variables
export PROJECT_ID=$(gcloud config get-value project)
export IMAGE_NAME=hello-gke
export IMAGE_TAG=latest
export REGISTRY=gcr.io

# 2.2 Build Docker image for AMD64 (GKE compatible architecture)
docker buildx build --platform linux/amd64 -t ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG} --load .

# Alternative: Build without specifying platform (for local testing)
docker build -t ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG} .

# 2.3 Verify the image was built
docker images | grep hello-gke

# 2.4 Test the image locally (optional)
docker run -d -p 8080:8080 --name hello-gke-test ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG}
sleep 5
curl http://localhost:8080/
docker stop hello-gke-test
docker rm hello-gke-test


================================================================================
SECTION 3: PUSH IMAGE TO GOOGLE CONTAINER REGISTRY
================================================================================

# 3.1 Push image to GCR
docker push ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG}

# 3.2 Verify the image in GCR
gcloud container images list --filter="name:hello-gke"

# 3.3 View image details in GCR
gcloud container images describe ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG}

# 3.4 Tag with version number (optional - recommended for production)
export VERSION=1.0.0
docker tag ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG} ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${VERSION}
docker push ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${VERSION}


================================================================================
SECTION 4: CREATE GKE CLUSTER (if needed)
================================================================================

# 4.1 Create a new GKE cluster
export CLUSTER_NAME=hello-gke-cluster
export ZONE=us-central1-a
export REGION=us-central1
export NUM_NODES=3

gcloud container clusters create ${CLUSTER_NAME} \
  --zone ${ZONE} \
  --num-nodes ${NUM_NODES} \
  --machine-type n1-standard-1 \
  --enable-stackdriver-kubernetes \
  --logging=SYSTEM_COMPONENTS,WORKLOADS \
  --monitoring=SYSTEM_COMPONENTS,WORKLOADS

# 4.2 List all clusters
gcloud container clusters list

# 4.3 Describe a specific cluster
gcloud container clusters describe ${CLUSTER_NAME} --zone ${ZONE}

SECTION 6: UPDATE DEPLOYMENT MANIFEST
================================================================================

# 6.1 Update k8s/deployment.yaml with your image reference
# Edit line 22 in k8s/deployment.yaml:

SECTION 7: DEPLOY APPLICATION TO GKE
================================================================================

# 7.1 Create dev namespace if not exists
kubectl create namespace dev --dry-run=client -o yaml | kubectl apply -f -

# 7.2 Deploy all resources to dev namespace (Deployment, Service, ConfigMap, ServiceAccount)

# 7.3 Verify deployment was created

# 7.4 Check pods status

# 7.5 Watch pods being created
kubectl get pods -l app=hello-gke -n dev --watch

# 7.6 Get detailed pod information
kubectl describe pods -l app=hello-gke -n dev

# 7.7 Check services
kubectl get svc hello-gke -n dev

# 7.8 Watch for LoadBalancer external IP (may take 1-2 minutes)
kubectl get svc hello-gke -n dev --watch

SECTION 15: IMPORTANT NOTES

⚠️  BEFORE DEPLOYMENT:
- Replace YOUR_PROJECT_ID with your actual GCP project ID
- Ensure you have a GKE cluster already created, or create one using Section 4
- Update k8s/deployment.yaml with the correct image reference (Section 6)

✅  BEST PRACTICES:
- Always build for amd64 platform (GKE compatible)
- Use semantic versioning for image tags (v1.0.0, v1.0.1, etc.)
- Use ConfigMaps for configuration instead of hardcoding
- Set resource requests and limits in deployment
- Enable logging and monitoring in GKE cluster
- Use namespaces to organize resources
- Implement health checks and probes

🔒  SECURITY:
- Use non-root user in Docker image (already configured)
- Enable Pod Security Policies in GKE
- Use RBAC for access control
- Enable network policies
- Scan images for vulnerabilities
- Use private image registries
- Rotate credentials regularly

📊  MONITORING:
- Enable Cloud Logging and Cloud Monitoring
- Set up alerts for critical metrics
- Monitor pod CPU and memory usage
- Check deployment health regularly

SECTION 2: BUILD DOCKER IMAGE
================================================================================

# 2.1 Set environment variables
export PROJECT_ID=$(gcloud config get-value project)
export IMAGE_NAME=hello-gke
export IMAGE_TAG=latest
export REGISTRY=gcr.io

# 2.2 Build Docker image for AMD64 (GKE compatible architecture)
docker buildx build --platform linux/amd64 -t ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG} --load .

# Alternative: Build without specifying platform (for local testing)
docker build -t ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG} .

# 2.3 Verify the image was built
docker images | grep hello-gke

# 2.4 Test the image locally (optional)
docker run -d -p 8080:8080 --name hello-gke-test ${REGISTRY}/${PROJECT_ID}/${IMAGE_NAME}:${IMAGE_TAG}
sleep 5
curl http://localhost:8080/
docker stop hello-gke-test
docker rm hello-gke-test

# 24.1 Validate Helm chart

# 24.2 Preview Helm manifests
helm template hello-gke helm/hello-gke/
helm template hello-gke helm/hello-gke/ --values helm/hello-gke/values-dev.yaml

# 24.3 Dry-run Helm installation
helm install hello-gke helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set global.projectId=${PROJECT_ID} \
  --dry-run --debug

# 24.4 Create dev namespace if not exists
kubectl create namespace dev --dry-run=client -o yaml | kubectl apply -f -

# 24.5 Install Helm release to dev namespace
helm install hello-gke helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set global.projectId=${PROJECT_ID} \
  --wait \
  --timeout 5m

# 24.6 Install Helm release with custom image tag
helm install hello-gke helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set global.projectId=${PROJECT_ID} \
  --set image.tag=v1.0.0 \
  --set image.pullPolicy=Always \
  --wait

# 24.7 List Helm releases
helm list
helm list -n dev
helm list -a -A

# 24.8 Get Helm release status
helm status hello-gke -n dev

# 24.9 Get release notes
helm get notes hello-gke -n dev

# 24.10 Get release values
helm get values hello-gke -n dev

# 24.11 Get generated manifests
helm get manifest hello-gke -n dev

# 24.12 Upgrade Helm release (update image)
helm upgrade hello-gke helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set global.projectId=${PROJECT_ID} \
  --set image.tag=v1.0.1 \
  --wait

# 24.13 Upgrade Helm release (update configuration)
helm upgrade hello-gke helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set app.replicaCount=3 \
  --set app.properties.logLevel=DEBUG \
  --wait

# 24.14 View Helm release history
helm history hello-gke -n dev

# 24.15 Rollback Helm release to previous version
helm rollback hello-gke -n dev
helm rollback hello-gke 1 -n dev

# 24.16 Verify rollback
helm status hello-gke -n dev
helm history hello-gke -n dev

# 24.17 Deploy to production with Helm
helm install hello-gke-prod helm/hello-gke/ \
  --namespace prod \
  --values helm/hello-gke/values-prod.yaml \
  --set global.projectId=${PROJECT_ID} \
  --set image.tag=v1.0.0 \
  --wait \
  --timeout 5m

# 24.18 Deploy to multiple environments
# Dev
helm install hello-gke-dev helm/hello-gke/ \
  --namespace dev \
  --values helm/hello-gke/values-dev.yaml \
  --set global.projectId=${PROJECT_ID}

# Staging
helm install hello-gke-staging helm/hello-gke/ \
  --namespace staging \
  --values helm/hello-gke/values.yaml \
  --set global.projectId=${PROJECT_ID}

# Prod
helm install hello-gke-prod helm/hello-gke/ \
  --namespace prod \
  --values helm/hello-gke/values-prod.yaml \
  --set global.projectId=${PROJECT_ID}

# 24.19 Check all Helm releases across namespaces
helm list -A

# 24.20 Get status of all releases
helm list -A | grep hello-gke

# 24.21 Uninstall Helm release
helm uninstall hello-gke -n dev

# 24.22 Uninstall release from all environments
helm uninstall hello-gke-dev -n dev
helm uninstall hello-gke-staging -n staging
helm uninstall hello-gke-prod -n prod

# 24.23 Uninstall and delete namespace
helm uninstall hello-gke -n dev
kubectl delete namespace dev
